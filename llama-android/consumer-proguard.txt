# Consumer R8/ProGuard rules — shipped as proguard.txt inside the AAR, so AGP
# applies them to every consuming app automatically (no manual rule copying,
# unlike the source-integration flow documented in the core README).

# The native layer resolves classes, fields, and methods by name:
# JNI_OnLoad calls FindClass/GetMethodID/GetFieldID on net.ladenthin.llama
# types (LlamaModel, exception.LlamaException, value.LogLevel, args.LogFormat,
# callback.LoadProgressCallback, ...), and Jackson reflects over the parameter
# and response POJOs. Keep the whole binding.
-keep class net.ladenthin.llama.** { *; }

# Native method names must survive for JNI registration.
-keepclasseswithmembernames class * {
    native <methods>;
}

# Jackson databind needs generic signatures and annotations at runtime.
-keepattributes Signature,InnerClasses,EnclosingMethod,*Annotation*
-dontwarn com.fasterxml.jackson.databind.**

# JVM-only optional integrations referenced from the core classes but absent
# (and unused) on Android.
-dontwarn org.slf4j.**

# Compile-time-only references R8 cannot resolve on Android. The broad keep above
# retains the whole binding, so R8 verifies every referenced type:
# - com.sun.net.httpserver: the JDK's HTTP server, used only by the JVM-only
#   OpenAiCompatServer transport. Those classes stay dexed but are unusable on
#   Android by design (loading them throws NoClassDefFoundError; the supported
#   Android entry points never touch them).
# - lombok / animal-sniffer: CLASS-retention build-time annotations referenced
#   from the bytecode; they exist on no runtime classpath anywhere.
# - org.checkerframework: the Nullness Checker writes its qualifiers (@NonNull,
#   @Initialized, @Pure, ...) into the class files it checks. They are RUNTIME
#   retention, but checker-qual is provided scope in the core and not in the AAR
#   pom (major 55 classes), so no app has it; a runtime that reflects over an
#   annotated element skips an annotation type it cannot resolve.
-dontwarn com.sun.net.httpserver.**
-dontwarn lombok.**
-dontwarn org.codehaus.mojo.animal_sniffer.**
-dontwarn org.checkerframework.**
